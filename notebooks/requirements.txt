# Lab 3a retrieval harness.
#
# chromadb and numpy are PINNED because they determine index semantics and float
# behaviour: chromadb 1.x changed how the HNSW distance space is configured, and getting
# that wrong silently leaves you on squared-L2 while every label still reads "cosine".
# The rest are floored — they do not affect results.
#
# Verified 2026-10-14 on Homebrew Python 3.14.7 (macOS arm64): resolves cleanly with the
# full set recorded in requirements.lock.txt, all prebuilt wheels, no source builds. The
# only other interpreter here is /usr/bin/python3 (3.9.6), which is not viable.
#
# chromadb pulls onnxruntime (~100MB) for its default embedding function, which this
# notebook never uses. It cannot be dropped — and its presence is exactly what arms the
# `query_texts=` trap the notebook warns about.
chromadb==1.5.9
numpy==2.5.4
httpx>=0.27
jupyterlab>=4
ipykernel>=6

# Lab 1 (agent/) — the HR RAG assistant and its OpenAI-compatible server.
# fastapi is pinned because the served response shapes are part of the harness contract:
# SPLX calls /v1/responses and a change in how FastAPI serialises that body is a change in
# what the grader reads. Resolved 2026-10-14 to fastapi 0.143.1 + starlette 1.8.0, both
# pure-python wheels on Python 3.14.7, no other dependency changes.
fastapi==0.143.1
# uvicorn and python-dotenv arrive transitively via chromadb, but agent/ imports them
# directly. A transitive dependency that a later chromadb release drops would break the
# server with an unrelated-looking ImportError, so they are explicit.
uvicorn>=0.54
python-dotenv>=1.2
